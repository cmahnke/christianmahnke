{{- /* Print rendering: fetch the post image and caption at build time so Vivliostyle never has to run Instagram's embed script. */ -}} {{- $id := .id -}} {{- $captionOverride := .caption | default "" -}} {{- $permalink := printf "https://www.instagram.com/p/%s/" $id -}} {{- $embedHTML := "" -}} {{- $embed := try (resources.GetRemote (printf "https://www.instagram.com/p/%s/embed/captioned/" $id)) -}} {{- with $embed.Value -}} {{- $embedHTML = .Content -}} {{- end -}} {{- $caption := $captionOverride -}} {{- $extracted := false -}} {{- if eq $caption "" -}} {{- $captionMatches := findRE `(?s)
(.*?)
` $embedHTML -}} {{- if gt (len $captionMatches) 0 -}} {{- $caption = index $captionMatches 0 | replaceRE `<[^>]+>` " " | replaceRE `\s+` " " | plainify | strings.TrimSpace -}} {{- $extracted = ne $caption "" -}} {{- end -}} {{- end -}} {{- if eq $caption "" -}} {{- $caption = $permalink -}} {{- end -}} {{- $image := "" -}} {{- $remote := try (resources.GetRemote (printf "https://www.instagram.com/p/%s/media/?size=l" $id)) -}} {{- with $remote.Value -}} {{- if and (eq .MediaType.MainType "image") (gt .Width 200) -}} {{- $image = . -}} {{- end -}} {{- end -}} {{- if not $image -}} {{- $imgTags := findRE `]*>` $embedHTML -}} {{- if gt (len $imgTags) 0 -}} {{- $src := replaceRE `(?s)^.*src="([^"]+)".*$` `$1` (index $imgTags 0) -}} {{- $src = replaceRE `&` `&` $src -}} {{- $fallback := try (resources.GetRemote $src) -}} {{- with $fallback.Value -}} {{- if and (eq .MediaType.MainType "image") (gt .Width 200) -}} {{- $image = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
